SELECT EMPLOYEE_ID,
       FIRST_NAME,
       LAST_NAME,
       JOB_ID,
       DEPARTMENT_ID,
       SALARY                AS Employee_Salary,
       ROUND((SELECT AVG(SALARY)
       FROM EMPLOYEES), 2) AS Company_Avg,
       ROUND(SALARY - 
       (SELECT AVG(SALARY) 
       FROM EMPLOYEES), 2) AS Above_Avg_By
       FROM EMPLOYEES
       WHERE SALARY > (SELECT AVG(SALARY)
       FROM EMPLOYEES)
       ORDER BY SALARY DESC;
